Script de redirecionamento do pop-up

Depois do cadastro no pop-up, a pessoa vai sozinha para o checkout da HeroSpark levando a origem da campanha (UTM). É o mesmo código que já está no ar e testado na v8, v9 e v10.

Jeito mais fácil:

Duplique a imersao-pratica-v9 e troque só textos e imagens. O pop-up, o envio para o ActiveCampaign, o rastreio e o redirecionamento já vêm prontos. Se a página nova já tem outro layout, use os blocos abaixo.

Opção 1 · só o redirecionamento

Esta função leva a pessoa ao checkout e repassa as UTMs da página. Chame irParaCheckout() depois que o cadastro for enviado.

Bloco 1

Função de redirecionamento

copiar
  var redirecionado=false;
  function irParaCheckout(){
    if(redirecionado) return; redirecionado=true;
    // leva a UTM da página pro checkout; o GTM completa src/sck no clique
    var url=CHECKOUT;
    if(location.search) url+=(url.indexOf('?')<0?'?':'&')+location.search.slice(1);
    var a=document.createElement('a'); a.href=url; a.style.display='none';
    document.body.appendChild(a); a.click();
  }

Opção 2 · script inteiro do pop-up (igual à v9)

Abre o pop-up, valida nome, e-mail e telefone, conta o Lead uma vez só, envia para o ActiveCampaign e redireciona. Cole no fim da página, antes de </body>, no lugar do script atual do pop-up. Se a página não tiver o carrossel, essa parte não atrapalha: ela só roda quando acha o carrossel.

Bloco 2

Script completo do pop-up

copiar
<script>
(function(){
  /* ---------- Carrossel de depoimentos ---------- */
  var t=document.getElementById('ibpm-track');
  if(t){
    var w=t.closest('.carousel');
    function step(){var i=t.querySelector('img');return i?i.offsetWidth+20:300;}
    w.querySelector('.prev').addEventListener('click',function(){t.scrollBy({left:-step()*2,behavior:'smooth'});});
    w.querySelector('.next').addEventListener('click',function(){t.scrollBy({left:step()*2,behavior:'smooth'});});
  }

  /* ---------- Popup de cadastro ---------- */
  var CHECKOUT = 'https://pay.herospark.com/imersao-pratica-botox-com-seguranca-medica-ingresso-comum-526980';
  var ENDPOINT = 'ac-imersao-botox.php';    // arquivo PHP na MESMA pasta deste HTML
  var TIMEOUT  = 8000;                      // ms máximos esperando o ActiveCampaign

  var pop=document.getElementById('ibpm-pop');
  var form=document.getElementById('ibpm-form');
  if(!pop||!form) return;

  var btn=form.querySelector('.pop-btn');
  var tel=document.getElementById('ibpm-tel');
  var lastFocus=null;

  function rastrear(evento,dados){
    try{ if(window.fbq) fbq(evento.custom?'trackCustom':'track',evento.fb,dados.fb||{}); }catch(e){}
    try{ if(window.gtag) gtag('event',evento.ga,dados.ga||{}); }catch(e){}
    try{ (window.dataLayer=window.dataLayer||[]).push(Object.assign({event:evento.dl},dados.dl||{})); }catch(e){}
  }

  var popupRastreado=false;
  function abrir(){
    if(!popupRastreado){
      popupRastreado=true;
      rastrear({fb:'PopupAberto',custom:true,ga:'popup_aberto',dl:'popup_aberto'},
        {fb:{content_name:'Imersao Botox na Pratica Medica'},
         ga:{}});
    }
    lastFocus=document.activeElement;
    pop.classList.add('is-open');
    pop.setAttribute('aria-hidden','false');
    document.documentElement.classList.add('ibpm-lock');
    setTimeout(function(){document.getElementById('ibpm-nome').focus();},120);
  }
  function fechar(){
    pop.classList.remove('is-open');
    pop.setAttribute('aria-hidden','true');
    document.documentElement.classList.remove('ibpm-lock');
    if(lastFocus&&lastFocus.focus) lastFocus.focus();
  }

  document.querySelectorAll('[data-ibpm-pop], a[href="#popup-ingresso"]').forEach(function(a){
    a.addEventListener('click',function(e){e.preventDefault();abrir();});
  });
  pop.querySelector('.pop-close').addEventListener('click',fechar);
  pop.addEventListener('click',function(e){if(e.target===pop) fechar();});
  document.addEventListener('keydown',function(e){if(e.key==='Escape'&&pop.classList.contains('is-open')) fechar();});

  // Máscara de telefone (00) 00000-0000
  tel.addEventListener('input',function(){
    var d=tel.value.replace(/\D/g,'').slice(0,11), r=d;
    if(d.length>10)      r='('+d.slice(0,2)+') '+d.slice(2,7)+'-'+d.slice(7);
    else if(d.length>6)  r='('+d.slice(0,2)+') '+d.slice(2,6)+'-'+d.slice(6);
    else if(d.length>2)  r='('+d.slice(0,2)+') '+d.slice(2);
    else if(d.length>0)  r='('+d;
    tel.value=r;
  });

  function setErro(input,erro){input.closest('.pop-field').classList.toggle('has-error',erro);}
  form.querySelectorAll('input').forEach(function(i){
    i.addEventListener('input',function(){if(i.closest('.pop-field')) setErro(i,false);});
  });

  function validar(){
    var nome=document.getElementById('ibpm-nome'), email=document.getElementById('ibpm-email');
    var ok=true, digitos=tel.value.replace(/\D/g,'');
    if(nome.value.trim().split(/\s+/).length<2){setErro(nome,true);ok=false;}
    if(!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email.value.trim())){setErro(email,true);ok=false;}
    if(digitos.length<10||digitos.length>11){setErro(tel,true);ok=false;}
    if(!ok){var f=form.querySelector('.has-error input'); if(f) f.focus();}
    return ok;
  }

  var redirecionado=false;
  function irParaCheckout(){
    if(redirecionado) return; redirecionado=true;
    // leva a UTM da página pro checkout; o GTM completa src/sck no clique
    var url=CHECKOUT;
    if(location.search) url+=(url.indexOf('?')<0?'?':'&')+location.search.slice(1);
    var a=document.createElement('a'); a.href=url; a.style.display='none';
    document.body.appendChild(a); a.click();
  }

  form.addEventListener('submit',function(e){
    e.preventDefault();
    if(btn.disabled||!validar()) return;
    btn.disabled=true; btn.textContent='ENVIANDO...';

    rastrear({fb:'Lead',ga:'generate_lead',dl:'lead_popup_imersao'},
      {fb:{value:97,currency:'BRL',content_name:'Imersao Botox na Pratica Medica'},
       ga:{value:97,currency:'BRL',form_id:'popup_imersao'}});

    var fd=new FormData(form);
    var ctrl=('AbortController' in window)?new AbortController():null;
    var timer=setTimeout(function(){if(ctrl) ctrl.abort(); irParaCheckout();},TIMEOUT);

    fetch(ENDPOINT,{method:'POST',body:fd,credentials:'same-origin',signal:ctrl?ctrl.signal:undefined})
      .catch(function(){ /* mesmo com falha, segue para o checkout para não perder a venda */ })
      .then(function(){clearTimeout(timer); irParaCheckout();});
  });
})();
</script>

Antes de publicar, confira

  1. CHECKOUT com o link certo da HeroSpark para esta página (hoje: ingresso comum …-526980).
  2. O ac-imersao-botox.php na mesma pasta do HTML, porque o ENDPOINT procura por ele lá.
  3. Mantenha os IDs do pop-up iguais aos da v9: ibpm-pop, ibpm-form, ibpm-nome, ibpm-email, ibpm-tel. O Google Tag Manager usa esses nomes para não contar o Lead em dobro.
  4. O botão que abre o pop-up precisa ter data-ibpm-pop ou href="#popup-ingresso".
  5. Mantenha o pixel do Meta e o GTM no topo da página, iguais aos da v9.

Teste rápido

  1. Abra a página com ?utm_source=teste&utm_campaign=teste no fim do endereço.
  2. Abra o pop-up, preencha com dados de teste e envie.
  3. No endereço do checkout precisam aparecer utm_source=teste, utm_campaign=teste, src= e sck=.
  4. Mande o link da página nova para a GGD, que a gente confere o pixel e o GA4. Depois apague o contato de teste no ActiveCampaign.
